4. Write a C program using Unix system calls to open a file for writing (create it if it does not exist). The program should read text from the keyboard and write it into file until Ctrl+D.

#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>
int main() {
    int fd, n;
    char fname[20], buffer[1000];
    printf("Enter filename: ");
    scanf("%s", fname);
    fd = open(fname, O_CREAT | O_WRONLY | O_TRUNC, 0644);
    if (fd == -1) {
        printf("Open error");
        return 0;
    }
    printf("Enter the file contents to be written:\n");
    while (1) {
        n = read(0, buffer, sizeof(buffer));
        if (n > 0)
            write(fd, buffer, n);
        else if (n == 0)
            break;
        else {
            printf("Read error");
            close(fd);
            return 0;
        }
    }
    close(fd);
    printf("Contents are written to file");
    return 0;
}
